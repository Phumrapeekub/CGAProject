@admin_bp.get("/assessments")
def assessments_list():
    if not _require_admin(): return redirect(url_for("auth.login"))
    q = (request.args.get("q") or "").strip()
    page = int(request.args.get("page", 1))
    per_page = 20
    offset = (page - 1) * per_page
    assesses = []
    total_count = 0
    
    # --- 1. TRY SUPABASE ---
    try:
        supabase_url = os.getenv("SUPABASE_URL")
        supabase_key = os.getenv("SUPABASE_KEY")
        if supabase_url and supabase_key:
            supabase = create_client(supabase_url, supabase_key)
            res = supabase.table("cga_records").select("*").order("id", desc=True).limit(100).execute()
            if res.data:
                for r in res.data:
                    if q and q.lower() not in (r.get("full_name") or "").lower() and q.lower() not in (r.get("hn") or "").lower(): continue
                    assesses.append({
                        "header_id": r.get("id"), "assessment_date": r.get("assessed_date") or r.get("created_at", "")[:10],
                        "hn": r.get("hn"), "full_name": r.get("full_name"),
                        "scores": [{"instrument": "MMSE", "total_score": r.get("mmse_score")}, {"instrument": "TGDS", "total_score": r.get("tgds_score")}],
                        "status_label": "สมบูรณ์", "status_color": "green"
                    })
    except: pass

    # --- 2. TRY MYSQL FALLBACK ---
    if not assesses:
        try:
            conn = get_db_connection()
            if conn:
                cur = conn.cursor(dictionary=True)
                cur.execute("SHOW TABLES LIKE 'stg_cga_csv'")
                if cur.fetchone():
                    sql = "SELECT * FROM stg_cga_csv"
                    if q: sql += " WHERE first_name LIKE %s OR last_name LIKE %s OR hn LIKE %s"
                    sql += " ORDER BY id DESC LIMIT 50"
                    cur.execute(sql, (f"%{q}%", f"%{q}%", f"%{q}%") if q else ())
                    for r in cur.fetchall():
                        assesses.append({
                            "header_id": r["id"], "assessment_date": r["assessed_date_text"] or "-",
                            "hn": r["hn"], "full_name": f"{r.get('first_name','')} {r.get('last_name','')}",
                            "scores": [{"instrument": "MMSE", "total_score": r["mmse_score"]}],
                            "status_label": "สมบูรณ์", "status_color": "green"
                        })
                cur.close(); conn.close()
        except: pass

    total_count = len(assesses)
    total_pages = max((total_count + per_page - 1) // per_page, 1)
    return render_template("admin/assessments.html", assessments=assesses, active_page="assessments", q=q, page=page, total_pages=total_pages, total_count=total_count, target_patient_id=None, start_date=None, end_date=None)
